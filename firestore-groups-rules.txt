// ===== قواعد المجموعات الجديدة (مدير، نقل ملكية، دعوات) =====
// داخل:  match /chats/{chatId} {
// استبدل أسطر  allow create  و  allow update  و  allow delete  الحالية بهذي (ولا تلمس match /messages وغيرها)

      allow create: if signedIn() && (
        (request.resource.data.get('isGroup', false) == true
          && request.resource.data.owner == request.auth.uid
          && request.auth.uid in request.resource.data.members
          && request.resource.data.members.size() >= 1
          && request.resource.data.members.size() <= 50
          && request.resource.data.name is string
          && request.resource.data.name.size() <= 40
          && request.resource.data.get('admins', []).size() == 0
          && request.resource.data.get('invites', {}).size() <= 49)
        || (request.resource.data.get('isGroup', false) != true
          && request.resource.data.members.size() == 2
          && request.auth.uid in request.resource.data.members
          && !blockedBy(request.resource.data.members[0])
          && !blockedBy(request.resource.data.members[1])));

      allow update: if signedIn() && (
        (resource.data.get('isGroup', false) == true && (
          // المالك: كل شي، والمالك الجديد لازم يكون عضو
          (resource.data.owner == request.auth.uid
            && request.resource.data.isGroup == true
            && request.resource.data.owner in request.resource.data.members
            && request.resource.data.members.size() <= 50
            && request.resource.data.name.size() <= 40)
          // المدير: الاسم والأعضاء والدعوات، بدون تغيير المالك أو المدراء، وما يشيل المالك
          || (request.auth.uid in resource.data.get('admins', [])
            && request.auth.uid in resource.data.members
            && !request.resource.data.diff(resource.data).affectedKeys().hasAny(['owner', 'admins', 'isGroup'])
            && resource.data.owner in request.resource.data.members
            && request.resource.data.members.size() <= 50
            && request.resource.data.name.size() <= 40)
          // أي عضو: آخر رسالة، القراءة، يكتب الآن
          || (request.auth.uid in resource.data.members
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['lastMsg', 'lastTs', 'lastFrom', 'lastKind', 'read', 'typing']))
          // عضو (غير المالك) يغادر
          || (request.auth.uid in resource.data.members
            && resource.data.owner != request.auth.uid
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['members', 'admins'])
            && request.resource.data.members.toSet() == resource.data.members.toSet().difference([request.auth.uid].toSet())
            && !(request.auth.uid in request.resource.data.get('admins', [])))
          // المدعو: يقبل (يضيف نفسه فقط) أو يرفض (يشيل دعوته فقط)
          || (request.auth.uid in resource.data.get('invites', {})
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['members', 'invites'])
            && request.resource.data.get('invites', {}).diff(resource.data.get('invites', {})).affectedKeys().hasOnly([request.auth.uid])
            && !(request.auth.uid in request.resource.data.get('invites', {}))
            && (request.resource.data.members == resource.data.members
              || request.resource.data.members.toSet() == resource.data.members.toSet().union([request.auth.uid].toSet()))
            && request.resource.data.members.size() <= 50)
        ))
        || (resource.data.get('isGroup', false) != true
          && request.auth.uid in resource.data.members
          && !blockedBy(resource.data.members[0])
          && !blockedBy(resource.data.members[1])));

      allow delete: if signedIn()
        && resource.data.get('isGroup', false) == true
        && resource.data.owner == request.auth.uid;
